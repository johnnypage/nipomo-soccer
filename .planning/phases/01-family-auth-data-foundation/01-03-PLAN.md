**Create client/src/hooks/use-auth.tsx:**

```typescript
import { useQuery, useQueryClient } from "@tanstack/react-query";
import { getQueryFn, apiRequest } from "@/lib/queryClient";
import type { Family, Kid } from "@shared/schema";

interface AuthData {
  family: Pick<Family, "id" | "email" | "name" | "isRegistered" | "createdAt">;
  kids: Kid[];
}

export function useAuth() {
  const queryClient = useQueryClient();

  const { data, isLoading, error } = useQuery<AuthData | null>({
    queryKey: ["/api/auth/me"],
    queryFn: getQueryFn({ on401: "returnNull" }),
    staleTime: 5 * 60 * 1000, // 5 min (override default Infinity to allow refetch)
    retry: false,
  });

  const logout = async () => {
    await apiRequest("POST", "/api/auth/logout");
    queryClient.setQueryData(["/api/auth/me"], null);
    queryClient.invalidateQueries({ queryKey: ["/api/auth/me"] });
  };

  return {
    family: data?.family ?? null,
    kids: data?.kids ?? [],
    isLoading,
    isAuthenticated: !!data?.family,
    logout,
  };
}
```

Uses `getQueryFn({ on401: "returnNull" })` so unauthenticated users get null instead of an error (per PATTERNS.md queryClient pattern). The `credentials: "include"` is already set in getQueryFn, so session cookies are sent automatically (per RESEARCH Pitfall 7).

**Create client/src/hooks/use-active-kid.tsx:**

Following the exact CartProvider pattern from use-cart.tsx:

```typescript
import { createContext, useContext, useState, useEffect, useCallback, type ReactNode } from "react";
import { useAuth } from "./use-auth";
import type { Kid } from "@shared/schema";

interface ActiveKidContextType {
  activeKid: Kid | null;
  setActiveKidId: (kidId: string) => void;
  kids: Kid[];
}

const ActiveKidContext = createContext<ActiveKidContextType | null>(null);

export function ActiveKidProvider({ children }: { children: ReactNode }) {
  const { kids } = useAuth();
  const [activeKidId, setActiveKidId] = useState<string | null>(null);

  // Per D-08: Auto-select for single-kid families or when no selection
  useEffect(() => {
    if (kids.length > 0 && (!activeKidId || !kids.find(k => k.id === activeKidId))) {
      setActiveKidId(kids[0].id);
    }
    if (kids.length === 0) {
      setActiveKidId(null);
    }
  }, [kids, activeKidId]);

  const activeKid = kids.find(k => k.id === activeKidId) ?? null;

  return (
    <ActiveKidContext.Provider value={{ activeKid, setActiveKidId, kids }}>
      {children}
    </ActiveKidContext.Provider>
  );
}

export function useActiveKid() {
  const context = useContext(ActiveKidContext);
  if (!context) {
    throw new Error("useActiveKid must be used within an ActiveKidProvider");
  }
  return context;
}
```

Per D-09: Active kid stored in React context (client-side state).
Per D-08: Auto-selects for single-kid families.
Does NOT use localStorage (session-scoped, per PATTERNS.md note).

|----------|-------------|
| User input -> React forms | Email, kid names, birthdates entered by users |
| React forms -> Express API | Data sent via fetch with credentials: include |
| URL query params -> UI | ?error=expired param on signup page |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | use-auth.tsx | mitigate | Auth state derived from server-side session via GET /api/auth/me. Cannot be spoofed client-side -- 401 returns null. Redirect to /signup if unauthenticated. |
| T-03-02 | Tampering | SignupForm.tsx | mitigate | Client-side Zod validation with signupSchema before submission. Server re-validates (defense in depth). consentGiven must be true. |
| T-03-03 | Tampering | AddKidForm.tsx | mitigate | Client-side Zod validation with addKidSchema. Birthdate validated as YYYY-MM-DD regex. Server re-validates and computes age track server-side (not trusted from client). |
| T-03-04 | Information Disclosure | challenge/index.tsx | mitigate | Only displays kid data from the authenticated family's own session. Kids list comes from GET /api/auth/me which is scoped to session.familyId. |
| T-03-05 | Tampering | URL ?error=expired | accept | Only used to show a user-facing message. No state mutation from query params. XSS not possible -- value is checked as exact string match, not rendered as HTML. |
</threat_model>
